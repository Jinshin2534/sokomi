`src/lib/homography.js`:
```js
// 4点の対応から射影変換 (3x3, 行優先 9要素) を求める。
// 点の順番は [つま先, かかと, 内側の最大幅, 外側の最大幅]。
export const SOLE_TARGETS = [
  [0.5, 0],
  [0.5, 1],
  [0, 0.35],
  [1, 0.35],
];

export function solveHomography(src, dst) {
  const A = [];
  const b = [];
  for (let i = 0; i < 4; i++) {
    const [x, y] = src[i];
    const [X, Y] = dst[i];
    A.push([x, y, 1, 0, 0, 0, -X * x, -X * y]);
    b.push(X);
    A.push([0, 0, 0, x, y, 1, -Y * x, -Y * y]);
    b.push(Y);
  }
  return [...solveLinear(A, b), 1];
}

// 部分ピボット付きガウス・ジョルダン消去
function solveLinear(A, b) {
  const n = b.length;
  const M = A.map((row, i) => [...row, b[i]]);
  for (let c = 0; c < n; c++) {
    let p = c;
    for (let r = c + 1; r < n; r++) if (Math.abs(M[r][c]) > Math.abs(M[p][c])) p = r;
    if (Math.abs(M[p][c]) < 1e-12) throw new Error('degenerate');
    [M[c], M[p]] = [M[p], M[c]];
    for (let r = 0; r < n; r++) {
      if (r === c) continue;
      const f = M[r][c] / M[c][c];
      for (let k = c; k <= n; k++) M[r][k] -= f * M[c][k];
    }
  }
  return M.map((row, i) => row[n] / row[i]);
}

export function applyHomography(H, [x, y]) {
  const w = H[6] * x + H[7] * y + H[8];
  return [(H[0] * x + H[1] * y + H[2]) / w, (H[3] * x + H[4] * y + H[5]) / w];
}

// 余因子行列 / 行列式。射影変換は定数倍に意味がないので正規化はしない。
export function invertHomography(H) {
  const [a, b, c, d, e, f, g, h, i] = H;
  const A = e * i - f * h;
  const B = -(d * i - f * g);
  const C = d * h - e * g;
  const det = a * A + b * B + c * C;
  if (det === 0 || !Number.isFinite(det)) throw new Error('degenerate');
  return [
    A, -(b * i - c * h), b * f - c * e,
    B, a * i - c * g, -(a * f - c * d),
    C, -(a * h - b * g), a * e - b * d,
  ].map((v) => v / det);
}

// 内側と外側の点が「つま先→かかと」の線の両側に、線の途中の高さで打たれているか。
export function checkPoints(points) {
  const [toe, heel, inner, outer] = points;
  const ax = heel[0] - toe[0];
  const ay = heel[1] - toe[1];
  const len2 = ax * ax + ay * ay;
  if (len2 < 1) throw new Error('degenerate');
  const side = (p) => (ax * (p[1] - toe[1]) - ay * (p[0] - toe[0])) / len2;
  const along = (p) => (ax * (p[0] - toe[0]) + ay * (p[1] - toe[1])) / len2;
  const si = side(inner);
  const so = side(outer);
  if (Math.abs(si) < 0.05 || Math.abs(so) < 0.05 || Math.sign(si) === Math.sign(so)) {
    throw new Error('degenerate');
  }
  for (const p of [inner, outer]) {
    const t = along(p);
    if (t < 0.05 || t > 0.95) throw new Error('degenerate');
  }
}

export function pointsToSole(points) {
  checkPoints(points);
  return solveHomography(points, SOLE_TARGETS);
}
```

```js
import { pointsToSole, invertHomography, applyHomography } from './homography.js';

export const WARP_W = 120;
export const WARP_H = 300;
const SUB = 3; // 1画素あたり 3×3 の副標本で平均（縮小時のエイリアシング対策）

export function luminance(image) {
  const { width, height, data } = image;
  const out = new Float32Array(width * height);
  for (let i = 0, k = 0; i < out.length; i++, k += 4) {
    out[i] = 0.299 * data[k] + 0.587 * data[k + 1] + 0.114 * data[k + 2];
  }
  return { width, height, data: out };
}

// 画素中心は (px+0.5, py+0.5)。写真の外なら NaN。
function sampleBilinear(g, x, y) {
  const fx = x - 0.5;
  const fy = y - 0.5;
  const x0 = Math.floor(fx);
  const y0 = Math.floor(fy);
  if (x0 < 0 || y0 < 0 || x0 + 1 >= g.width || y0 + 1 >= g.height) return NaN;
  const tx = fx - x0;
  const ty = fy - y0;
  const i = y0 * g.width + x0;
  const d = g.data;
  const top = d[i] * (1 - tx) + d[i + 1] * tx;
  const bottom = d[i + g.width] * (1 - tx) + d[i + g.width + 1] * tx;
  return top * (1 - ty) + bottom * ty;
}

export function warpToSole(image, points, W = WARP_W, H = WARP_H) {
  const g = luminance(image);
  const inv = invertHomography(pointsToSole(points));
  const out = new Float32Array(W * H);
  for (let j = 0; j < H; j++) {
    for (let i = 0; i < W; i++) {
      let sum = 0;
      let bad = false;
      for (let sy = 0; sy < SUB && !bad; sy++) {
        for (let sx = 0; sx < SUB; sx++) {
          const u = (i + (sx + 0.5) / SUB) / W;
          const v = (j + (sy + 0.5) / SUB) / H;
          const [x, y] = applyHomography(inv, [u, v]);
          const s = sampleBilinear(g, x, y);
          if (Number.isNaN(s)) { bad = true; break; }
          sum += s;
        }
      }
      out[j * W + i] = bad ? NaN : sum / (SUB * SUB);
    }
  }
  return { width: W, height: H, data: out };
}
```

```js
// 照明のゲイン(明るさ・露出)とゆるい陰影を消すため、広い窓の局所平均で割る。
// 局所標準偏差では割らない: すり減って平らな所のノイズを増幅して、すり減りが見えなくなる。
export function illuminationNormalize(gray, radius = 15, floor = 8) {
  const { width: W, height: H, data } = gray;
  const S = new Float64Array((W + 1) * (H + 1));
  const N = new Float64Array((W + 1) * (H + 1));
  for (let y = 0; y < H; y++) {
    for (let x = 0; x < W; x++) {
      const v = data[y * W + x];
      const ok = !Number.isNaN(v);
      const k = (y + 1) * (W + 1) + (x + 1);
      S[k] = (ok ? v : 0) + S[k - 1] + S[k - (W + 1)] - S[k - (W + 1) - 1];
      N[k] = (ok ? 1 : 0) + N[k - 1] + N[k - (W + 1)] - N[k - (W + 1) - 1];
    }
  }
  const box = (A, x0, y0, x1, y1) =>
    A[y1 * (W + 1) + x1] - A[y0 * (W + 1) + x1] - A[y1 * (W + 1) + x0] + A[y0 * (W + 1) + x0];
  const out = new Float32Array(W * H);
  for (let y = 0; y < H; y++) {
    for (let x = 0; x < W; x++) {
      const v = data[y * W + x];
      if (Number.isNaN(v)) { out[y * W + x] = NaN; continue; }
      const x0 = Math.max(0, x - radius);
      const x1 = Math.min(W, x + radius + 1);
      const y0 = Math.max(0, y - radius);
      const y1 = Math.min(H, y + radius + 1);
      const mean = box(S, x0, y0, x1, y1) / box(N, x0, y0, x1, y1);
      out[y * W + x] = v / Math.max(mean, floor);
    }
  }
  return { width: W, height: H, data: out };
}
```

`src/lib/texture.js`:
```js
export const CELL = 10;

export function sobelMagnitude(img) {
  const { width: W, height: H, data: d } = img;
  const out = new Float32Array(W * H).fill(NaN);
  for (let y = 1; y < H - 1; y++) {
    for (let x = 1; x < W - 1; x++) {
      const i = y * W + x;
      const a = d[i - W - 1], b = d[i - W], c = d[i - W + 1];
      const l = d[i - 1], r = d[i + 1];
      const e = d[i + W - 1], f = d[i + W], g = d[i + W + 1];
      const gx = (c + 2 * r + g) - (a + 2 * l + e);
      const gy = (e + 2 * f + g) - (a + 2 * b + c);
      out[i] = Math.hypot(gx, gy); // NaN はそのまま伝わる
    }
  }
  return { width: W, height: H, data: out };
}

// マスごとの平均 = 模様の強さ。有効画素が 8割未満のマスは NaN。
export function cellTexture(mag, cell = CELL) {
  const { width: W, height: H, data } = mag;
  const cols = Math.floor(W / cell);
  const rows = Math.floor(H / cell);
  const values = new Float64Array(cols * rows);
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      let sum = 0;
      let n = 0;
      for (let y = r * cell; y < (r + 1) * cell; y++) {
        for (let x = c * cell; x < (c + 1) * cell; x++) {
          const v = data[y * W + x];
          if (!Number.isNaN(v)) { sum += v; n++; }
        }
      }
      values[r * cols + c] = n >= 0.8 * cell * cell ? sum / n : NaN;
    }
  }
  return { cols, rows, values };
}
```

```js
// 正規化座標 (u: 0=内側〜1=外側, v: 0=つま先〜1=かかと) での靴底の半幅。
const lerpCos = (a, b, t) => a + (b - a) * (1 - Math.cos(Math.PI * t)) / 2;

export function soleHalfWidth(v) {
  if (v <= 0 || v >= 1) return 0;
  if (v < 0.35) {
    const t = (0.35 - v) / 0.35;
    return 0.5 * Math.sqrt(1 - t * t);
  }
  if (v < 0.6) return lerpCos(0.5, 0.36, (v - 0.35) / 0.25);
  if (v < 0.85) return lerpCos(0.36, 0.38, (v - 0.6) / 0.25);
  const t = (v - 0.85) / 0.15;
  return 0.38 * Math.sqrt(1 - t * t);
}

export function insideSole(u, v, shrink = 1) {
  return Math.abs(u - 0.5) <= soleHalfWidth(v) * shrink;
}

// 靴底の縁と背景の境目は強いエッジだがすり減らない。混ぜないよう縮めた輪郭の内側だけ使う。
export const MASK_SHRINK = 0.9;

export function cellMask(cols, rows, shrink = MASK_SHRINK) {
  const out = new Array(cols * rows);
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      const u0 = c / cols, u1 = (c + 1) / cols;
      const v0 = r / rows, v1 = (r + 1) / rows;
      out[r * cols + c] = [[u0, v0], [u1, v0], [u0, v1], [u1, v1]].every(([u, v]) => insideSole(u, v, shrink));
    }
  }
  return out;
}
```

```js
// 新品時に模様がほとんど無いマスは比べようがないので判定不能にする。
// しきい値 = max(絶対下限, 靴底内の基準の模様の強さの中央値 × 比率)
export const MIN_TEXTURE_ABS = 0.15;
export const MIN_TEXTURE_REL = 0.25;

function median(xs) {
  if (xs.length === 0) return 0;
  const s = [...xs].sort((a, b) => a - b);
  const m = s.length >> 1;
  return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2;
}

export function cellWear(baseValues, nowValues, inSole) {
  const inside = [];
  for (let i = 0; i < baseValues.length; i++) {
    if (inSole[i] && Number.isFinite(baseValues[i])) inside.push(baseValues[i]);
  }
  const threshold = Math.max(MIN_TEXTURE_ABS, MIN_TEXTURE_REL * median(inside));
  return Array.from(baseValues, (base, i) => {
    const now = nowValues[i];
    if (!inSole[i] || !Number.isFinite(base) || !Number.isFinite(now) || base < threshold) return null;
    return Math.min(1, Math.max(0, 1 - now / base));
  });
}
```

```js
export const REGIONS = [
  { id: 'toe', label: 'つま先', test: (u, v) => v < 0.18 },
  { id: 'ball', label: '母趾球', test: (u, v) => v >= 0.18 && v < 0.42 && u < 0.5 },
  { id: 'outerBall', label: '小趾側', test: (u, v) => v >= 0.18 && v < 0.42 && u >= 0.5 },
  { id: 'arch', label: '土踏まず', test: (u, v) => v >= 0.42 && v < 0.68 },
  { id: 'heelIn', label: 'かかと内', test: (u, v) => v >= 0.68 && u < 0.5 },
  { id: 'heelOut', label: 'かかと外', test: (u, v) => v >= 0.68 && u >= 0.5 },
];

export const MIN_REGION_COVERAGE = 0.3;
export const MIN_OVERALL_COVERAGE = 0.25;

const mean = (xs) => xs.reduce((a, b) => a + b, 0) / xs.length;

// マスはすべて同じ面積なので、全体 = 判定可能マスの単純平均。
export function aggregate(wear, inSole, cols, rows) {
  const buckets = REGIONS.map(() => ({ inside: 0, values: [] }));
  const all = [];
  let insideTotal = 0;
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      const i = r * cols + c;
      if (!inSole[i]) continue;
      const u = (c + 0.5) / cols;
      const v = (r + 0.5) / rows;
      const b = buckets[REGIONS.findIndex((reg) => reg.test(u, v))];
      b.inside++;
      insideTotal++;
      if (wear[i] != null) {
        b.values.push(wear[i]);
        all.push(wear[i]);
      }
    }
  }
  const regions = REGIONS.map((reg, k) => {
    const { inside, values } = buckets[k];
    const coverage = inside ? values.length / inside : 0;
    return { id: reg.id, label: reg.label, mean: coverage >= MIN_REGION_COVERAGE ? mean(values) : null, coverage };
  });
  const overallCoverage = insideTotal ? all.length / insideTotal : 0;
  return { regions, overall: overallCoverage >= MIN_OVERALL_COVERAGE ? mean(all) : null, overallCoverage };
}
```

```js
import { pointsToSole, applyHomography } from './homography.js';
import { insideSole } from './mask.js';

export const DEMO_POINTS = [[200, 50], [200, 750], [60, 295], [340, 295]];

// 靴底を模した画像。v 方向に 30 本の溝(角の立った縞)。amp(u,v) は模様の残り具合 0〜1。
export function makeSoleImage({ width = 400, height = 800, points = DEMO_POINTS, amp = () => 1, gain = 1 } = {}) {
  const H = pointsToSole(points);
  const data = new Uint8ClampedArray(width * height * 4);
  const k4 = Math.tanh(4);
  for (let py = 0; py < height; py++) {
    for (let px = 0; px < width; px++) {
      const [u, v] = applyHomography(H, [px + 0.5, py + 0.5]);
      let val = 40;
      if (insideSole(u, v)) {
        const s = Math.tanh(4 * Math.sin(2 * Math.PI * v * 30)) / k4;
        val = 128 + 60 * amp(u, v) * s;
      }
      val *= gain;
      const k = (py * width + px) * 4;
      data[k] = data[k + 1] = data[k + 2] = val;
      data[k + 3] = 255;
    }
  }
  return { width, height, data };
}

export function transformPoints(points, { angle = 0, scale = 1, center = [200, 400], offset = [0, 0] } = {}) {
  const c = Math.cos(angle);
  const s = Math.sin(angle);
  return points.map(([x, y]) => {
    const dx = (x - center[0]) * scale;
    const dy = (y - center[1]) * scale;
    return [center[0] + offset[0] + dx * c - dy * s, center[1] + offset[1] + dx * s + dy * c];
  });
}

const clamp01 = (x) => Math.min(1, Math.max(0, x));

// デモ用: かかと外側ほど強く、つま先も少しすり減った靴
export function DEMO_WEAR(u, v) {
  const heel = clamp01((v - 0.6) / 0.4) * clamp01((u - 0.35) / 0.5);
  const toe = v < 0.15 ? 0.3 : 0;
  return clamp01(1 - 0.9 * heel - toe);
}
```

```js
import { insideSole, MASK_SHRINK } from './mask.js';

// 正規化前の靴底画像の統計。sharp = 平均|ラプラシアン| / 平均|勾配|。
// 模様の量には左右されにくく、エッジがぼけると下がる（すり減りとピント甘さを分けるため）。
export function photoStats(gray) {
  const { width: W, height: H, data: d } = gray;
  let sum = 0, n = 0, lap = 0, grad = 0, clipped = 0;
  for (let y = 1; y < H - 1; y++) {
    for (let x = 1; x < W - 1; x++) {
      if (!insideSole((x + 0.5) / W, (y + 0.5) / H, MASK_SHRINK)) continue;
      const i = y * W + x;
      const c = d[i], l = d[i - 1], r = d[i + 1], t = d[i - W], b = d[i + W];
      if ([c, l, r, t, b].some(Number.isNaN)) continue;
      sum += c;
      n++;
      if (c >= 250 || c <= 5) clipped++;
      lap += Math.abs(4 * c - l - r - t - b);
      grad += Math.hypot(r - l, b - t);
    }
  }
  return { mean: n ? sum / n : 0, sharp: grad > 0 ? lap / grad : 0, clipped: n ? clipped / n : 0 };
}

const LEVELS = ['高', '中', '低'];

export function compareQuality(baseGray, nowGray) {
  const base = photoStats(baseGray);
  const now = photoStats(nowGray);
  const reasons = [];
  let level = 0;
  const raise = (to, reason) => { level = Math.max(level, to); reasons.push(reason); };

  const bright = base.mean > 0 ? now.mean / base.mean : 1;
  if (bright < 0.5) raise(2, '今の写真が新品時よりかなり暗いです');
  else if (bright < 0.7) raise(1, '今の写真が新品時より少し暗いです');
  else if (bright > 2) raise(2, '今の写真が新品時よりかなり明るいです');
  else if (bright > 1.4) raise(1, '今の写真が新品時より少し明るいです');

  if (base.sharp > 0 && now.sharp > 0) {
    const sharp = now.sharp / base.sharp;
    if (sharp < 0.6) raise(2, 'ピントが甘いか、手ブレしています');
    else if (sharp < 0.8) raise(1, 'ややピントが甘いです');
  }

  if (now.clipped > 0.2) raise(1, '白飛び・黒つぶれが多いです');

  return { level: LEVELS[level], reasons };
}
```

```js
import { cellWear } from './wear.js';
import { aggregate } from './regions.js';
import { compareQuality } from './quality.js';

// 新品時の写真と今の写真(＋それぞれの4点)から、マス・部位・全体のすり減り度を出す。
// 点が不正なら Error('degenerate')。
export function analyze(baseImage, basePoints, nowImage, nowPoints) {
  const baseWarped = warpToSole(baseImage, basePoints);
  const nowWarped = warpToSole(nowImage, nowPoints);
  const texture = (g) => cellTexture(sobelMagnitude(illuminationNormalize(g)));
  const bt = texture(baseWarped);
  const nt = texture(nowWarped);
  const inSole = cellMask(bt.cols, bt.rows);
  const cells = cellWear(bt.values, nt.values, inSole);
  const { regions, overall, overallCoverage } = aggregate(cells, inSole, bt.cols, bt.rows);
  return {
    cols: bt.cols,
    rows: bt.rows,
    cells,
    inSole,
    regions,
    overall,
    overallCoverage,
    quality: compareQuality(baseWarped, nowWarped),
    nowWarped,
  };
}
```

```js
// IndexedDB の薄いラッパ。靴1足 = 1レコード（写真は Blob のまま入る）。
const DB_NAME = 'sokomi';
const STORE = 'shoes';

function openDb() {
  return new Promise((resolve, reject) => {
    const req = indexedDB.open(DB_NAME, 1);
    req.onupgradeneeded = () => req.result.createObjectStore(STORE, { keyPath: 'id' });
    req.onsuccess = () => resolve(req.result);
    req.onerror = () => reject(req.error);
  });
}

async function run(mode, fn) {
  const db = await openDb();
  return new Promise((resolve, reject) => {
    const tx = db.transaction(STORE, mode);
    const req = fn(tx.objectStore(STORE));
    tx.oncomplete = () => resolve(req?.result);
    tx.onerror = () => reject(tx.error);
  });
}

export async function listShoes() {
  const shoes = (await run('readonly', (s) => s.getAll())) ?? [];
  return shoes.sort((a, b) => b.createdAt - a.createdAt);
}
```

```js
import { h } from './dom.js';

export const POINT_LABELS = ['つま先の先端', 'かかとの先端', '内側(親指側)の一番広い所', '外側(小指側)の一番広い所'];
export const POINT_COLORS = ['#d9480f', '#1c7ed6', '#2b8a3e', '#e8590c'];

// 写真の上を順番にタップして4点を打つ。打った点はドラッグで直せる。
export function createPointPicker(photoCanvas, { onChange } = {}) {
  const canvas = h('canvas', { class: 'picker' });
  canvas.width = photoCanvas.width;
  canvas.height = photoCanvas.height;
  const ctx = canvas.getContext('2d');
  const hint = h('p', { class: 'picker-hint' });
  const points = [];
  let drag = -1;

  const copy = () => (points.length === 4 ? points.map((p) => [...p]) : null);
  const emit = () => onChange?.(copy());
  const toImage = (e) => {
    const r = canvas.getBoundingClientRect();
    return [((e.clientX - r.left) * canvas.width) / r.width, ((e.clientY - r.top) * canvas.height) / r.height];
  };
  const hitRadius = () => (canvas.width / canvas.getBoundingClientRect().width) * 24;

  function draw() {
    ctx.drawImage(photoCanvas, 0, 0);
    const rad = Math.max(6, canvas.width / 70);
    if (points.length >= 2) {
      ctx.save();
      ctx.setLineDash([rad, rad]);
      ctx.lineWidth = rad / 3;
      ctx.strokeStyle = 'rgba(255,255,255,0.9)';
      ctx.beginPath();
      ctx.moveTo(...points[0]);
      ctx.lineTo(...points[1]);
      ctx.stroke();
      ctx.restore();
    }
    points.forEach((p, i) => {
      ctx.beginPath();
      ctx.arc(p[0], p[1], rad, 0, Math.PI * 2);
      ctx.fillStyle = POINT_COLORS[i];
      ctx.fill();
      ctx.lineWidth = rad / 3;
      ctx.strokeStyle = '#fff';
      ctx.stroke();
      ctx.fillStyle = '#fff';
      ctx.font = `bold ${rad * 1.2}px sans-serif`;
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText(String(i + 1), p[0], p[1]);
    });
    hint.replaceChildren(
      points.length < 4
        ? h('span', {}, h('b', { style: `color:${POINT_COLORS[points.length]}` }, `${points.length + 1}. ${POINT_LABELS[points.length]}`), ' をタップ')
        : '4点そろいました。ずれていたら点をドラッグして直せます',
    );
  }

  canvas.addEventListener('pointerdown', (e) => {
    const p = toImage(e);
    const near = points.findIndex((q) => Math.hypot(q[0] - p[0], q[1] - p[1]) < hitRadius());
    if (near >= 0) {
      drag = near;
      canvas.setPointerCapture(e.pointerId);
    } else if (points.length < 4) {
      points.push(p);
      draw();
      emit();
    }
  });
  canvas.addEventListener('pointermove', (e) => {
    if (drag < 0) return;
    points[drag] = toImage(e);
    draw();
  });
  const endDrag = () => {
    if (drag < 0) return;
    drag = -1;
    emit();
  };
  canvas.addEventListener('pointerup', endDrag);
  canvas.addEventListener('pointercancel', endDrag);

  const reset = h('button', { class: 'ghost small', type: 'button', onClick: () => { points.length = 0; draw(); emit(); } }, '打ち直す');
  const legend = h('ol', { class: 'point-legend' },
    POINT_LABELS.map((l, i) => h('li', {}, h('i', { style: `background:${POINT_COLORS[i]}` }), l)));
  draw();
  return { el: h('div', { class: 'picker-wrap' }, hint, canvas, h('div', { class: 'picker-foot' }, legend, reset)), getPoints: copy };
}
```

```js
import { h } from './dom.js';
import { wearColor } from '../lib/color.js';
import { soleHalfWidth } from '../lib/mask.js';

const SCALE = 2;

function solePath(ctx, W, H) {
  ctx.beginPath();
  for (let k = 0; k <= 100; k++) {
    const v = k / 100;
    ctx.lineTo((0.5 - soleHalfWidth(v)) * W, v * H);
  }
  for (let k = 100; k >= 0; k--) {
    const v = k / 100;
    ctx.lineTo((0.5 + soleHalfWidth(v)) * W, v * H);
  }
  ctx.closePath();
}

function hatch(ctx, x, y, w, hgt) {
  ctx.save();
  ctx.beginPath();
  ctx.rect(x, y, w, hgt);
  ctx.clip();
  ctx.fillStyle = 'rgba(150,150,150,0.45)';
  ctx.fillRect(x, y, w, hgt);
  ctx.strokeStyle = 'rgba(90,90,90,0.6)';
  ctx.lineWidth = 1;
  for (let d = -hgt; d < w; d += 5) {
    ctx.beginPath();
    ctx.moveTo(x + d, y + hgt);
    ctx.lineTo(x + d + hgt, y);
    ctx.stroke();
  }
  ctx.restore();
}

// 今の写真を靴底座標にそろえたものを下地に、マスごとのすり減り度を重ねる。上=つま先、左=内側。
export function renderHeatmap(result) {
  const { nowWarped, cols, rows, cells, inSole } = result;
  const W = nowWarped.width * SCALE;
  const H = nowWarped.height * SCALE;
  const base = document.createElement('canvas');
  base.width = nowWarped.width;
  base.height = nowWarped.height;
  const bctx = base.getContext('2d');
  const img = bctx.createImageData(base.width, base.height);
  nowWarped.data.forEach((v, i) => {
    const g = Number.isNaN(v) ? 255 : v;
    img.data[i * 4] = img.data[i * 4 + 1] = img.data[i * 4 + 2] = g;
    img.data[i * 4 + 3] = 255;
  });
  bctx.putImageData(img, 0, 0);

  const canvas = h('canvas', { class: 'heatmap', 'aria-label': 'すり減りマップ' });
  canvas.width = W;
  canvas.height = H;
  const ctx = canvas.getContext('2d');
  ctx.drawImage(base, 0, 0, W, H);

  const cw = W / cols;
  const ch = H / rows;
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      const i = r * cols + c;
      if (!inSole[i]) continue;
      if (cells[i] == null) hatch(ctx, c * cw, r * ch, cw, ch);
      else {
        ctx.globalAlpha = 0.6;
        ctx.fillStyle = wearColor(cells[i]);
        ctx.fillRect(c * cw, r * ch, cw, ch);
        ctx.globalAlpha = 1;
      }
    }
  }

  // 靴底の外を白く薄める
  ctx.save();
  ctx.beginPath();
  ctx.rect(0, 0, W, H);
  solePath(ctx, W, H);
  ctx.fillStyle = 'rgba(255,255,255,0.75)';
  ctx.fill('evenodd');
  solePath(ctx, W, H);
  ctx.strokeStyle = 'rgba(60,50,40,0.5)';
  ctx.lineWidth = 2;
  ctx.stroke();
  ctx.restore();

  return h('figure', { class: 'heatmap-fig' },
    h('div', { class: 'heatmap-sides' }, h('span', {}, '内側'), h('span', {}, 'つま先 ↑'), h('span', {}, '外側')),
    canvas,
    h('div', { class: 'legend' },
      h('span', {}, '0%'), h('i', { class: 'legend-bar' }), h('span', {}, '100%'),
      h('i', { class: 'legend-hatch' }), h('span', {}, '判定不能')));
}
```

```markdown
# 底見 (sokomi)

靴底のすり減りを、新品のときに撮った写真と比べて推定するWebアプリ。

- 新品の靴底を撮って4点（つま先・かかと・内側と外側の一番広い所）をタップして登録
- あとで今の靴底を同じように撮ると、部位ごとのすり減りマップと全体の%が出る

## しくみ

1. 4点から射影変換を求め、2枚を「正規化した靴底座標」の 120×300 画像にそろえる
2. 広い窓の局所平均で割って、明るさ・露出の差を消す
3. Sobel 勾配の強さ（＝模様のくっきり具合）を 10×10px のマスごとに出す
4. すり減り度 = 1 − 今 / 新品時。新品時に模様が無いマスは判定不能
5. 6部位（つま先／母趾球／小趾側／土踏まず／かかと内／かかと外）と全体に集計し、明るさ・ピントの差から信頼度を出す

溝の深さを測っているわけではなく、模様の見え方の減り方から推定した値です。

## 開発

    pnpm install
    pnpm dev    # http://localhost:5730
    pnpm test

計算は `src/lib/` の純粋関数、画面は `src/ui/`。写真と記録はブラウザの IndexedDB にだけ保存されます。
```
